cmake_minimum_required(VERSION 3.25)
project(morphbyte_native LANGUAGES CXX)
add_library(morphbyte_native SHARED src/addon.cpp ${CMAKE_JS_SRC})
target_compile_features(morphbyte_native PRIVATE cxx_std_23)
target_compile_definitions(morphbyte_native PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=morphbyte_native)
target_include_directories(morphbyte_native PRIVATE ${CMAKE_JS_INC})
target_link_libraries(morphbyte_native PRIVATE ${CMAKE_JS_LIB})
set_target_properties(morphbyte_native PROPERTIES PREFIX "" SUFFIX ".node" OUTPUT_NAME "morphbyte")
if(MSVC)
  target_compile_options(morphbyte_native PRIVATE /W4 /EHsc)
else()
  target_compile_options(morphbyte_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
